        function procesarExcel(e) {
            const f = e.target.files[0]; if(!f) return;
            const r = new FileReader(); r.onload = ev => {
                try {
                    const wb = XLSX.read(new Uint8Array(ev.target.result), {type:'array'}), rows = XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]], {header:1});
                    let ag = 0, dp = 0;
                    for(let i=1; i<rows.length; i++){
                        if(!rows[i]||!rows[i][0]) continue;
                        const num = String(rows[i][0]).trim();
                        if(!db.colaboradores.find(c=>c.num===num)) {
                            db.colaboradores.push({ id: Date.now()+i, num, nombre: String(rows[i][1]||'').trim(), correo: String(rows[i][2]||'').trim(), placas: String(rows[i][3]||'').trim(), tc: String(rows[i][4]||'').trim(), foto_tc: null, placas_2: String(rows[i][5]||'').trim(), tc_2: String(rows[i][6]||'').trim(), foto_tc_2: null, prioridad: 0 });
                            ag++;
                        } else dp++;
                    }
                    if(ag>0) { guardarData(); mostrarToast(`${ag} importados (${dp} omitidos)`, "bg-green-500"); } else mostrarToast("Todos existían", "bg-yellow-500");
                } catch(err){ alert("Error de formato"); }
                document.getElementById('archivo-excel').value='';
            }; r.readAsArrayBuffer(f);
        }
